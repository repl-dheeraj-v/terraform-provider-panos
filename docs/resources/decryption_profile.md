# panos_decryption_profile (Resource)

Manages PAN-OS SSL/TLS decryption profiles, which define how the firewall inspects and handles decrypted traffic. Profiles can configure behavior for SSL forward proxy, SSL inbound proxy, SSL no-proxy, and protocol-level settings.

## Example Usage

```hcl
# Panorama shared decryption profile
resource "panos_decryption_profile" "shared_example" {
  location = {
    shared = {}
  }

  name        = "strict-tls-profile"
  description = "Enforce strong TLS settings"

  ssl_forward_proxy = {
    block_expired_certificate  = true
    block_untrusted_issuer     = true
    block_unknown_cert         = true
    block_unsupported_version  = true
    block_unsupported_cipher   = true
    restrict_cert_exts         = true
    strip_alpn                 = false
  }

  ssl_inbound_proxy = {
    block_unsupported_cipher        = true
    block_unsupported_version       = true
    block_tls13_downgrade_no_resource = false
  }

  ssl_protocol_settings = {
    min_version = "tls1-2"
    max_version = "max"
  }
}

# Device-group scoped decryption profile
resource "panos_decryption_profile" "dg_example" {
  location = {
    device_group = {
      panorama_device = "localhost.localdomain"
      name            = "shared"
    }
  }

  name = "inbound-inspection-profile"

  ssl_inbound_proxy = {
    block_if_no_resource   = true
    block_if_hsm_unavailable = false
  }

  ssl_protocol_settings = {
    min_version       = "tls1-2"
    max_version       = "max"
    keyxchg_algo_rsa  = true
    keyxchg_algo_ecdhe = true
  }
}
```
